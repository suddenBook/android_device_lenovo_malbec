# fastboot-info for malbec (TB390FU)
#
# Selected by TARGET_BOARD_FASTBOOT_INFO_FILE in BoardConfig.mk. Without it the
# build synthesises a default (build/make/core/Makefile:5929-5933) whose line 6
# is `flash recovery` -- three lines before `reboot fastboot`. That ordering is
# what bricked this device on the first flash attempt: it overwrites the recovery
# ramdisk and then immediately requires fastbootd, which lives in that ramdisk.
#
# The rule this file exists to enforce:
#
#   Never write the partition that implements the next step of your own procedure.
#
# recovery is not on the boot path at all -- boot.img carries no ramdisk,
# init_boot supplies the generic ramdisk and vendor_boot the vendor ramdisk.
# recovery serves only fastbootd and `adb sideload`.
#
# ⚠️ This file used to omit `flash recovery` entirely, on the grounds that the stock
# recovery is "the single in-band way back if our ROM does not boot". That premise
# is false and was never checked:
#
#   - the stock recovery ramdisk contains no reference to pstore at all
#     (`grep -rl pstore` over the extracted ramdisk returns 0 files), while ours
#     mounts /sys/fs/pstore at bootable/recovery/etc/init.rc:66. When the ROM does
#     not boot, the previous boot's kernel log is the only evidence there is, and
#     the stock recovery cannot read it;
#   - stock's recovery carries neither lib/modules nor an fstab; it takes both from
#     vendor_boot, which this ROM replaces. Keeping it produces an untested hybrid
#     rather than a known-good fallback;
#   - and the patched ramoops region is larger than stock's, so its base address
#     moves -- a boot on the stock DTB could not read our records even if it tried.
#
# The real way back is bootloader fastboot (ABL), which nothing here ever writes and
# which the volume-down hotkey reaches before the BCB is read.
#
# So recovery IS flashed -- as the very last line, after every logical partition and
# after `reboot fastboot`. The original hazard was flashing it *before* the step that
# needs fastbootd; position solves that, omission was never needed.
#
# NOTE: `fastboot flashall` on this device's artifacts will always take the
# reboot-into-fastbootd path, no matter what this file says. AOSP's fast path
# (OptimizedFlashSuperTask, system/core/fastboot/task.cpp:165-243) needs every
# partition in super_empty.img to carry LP_PARTITION_ATTR_READONLY
# (fs_mgr/liblp/super_layout_builder.cpp:53-56), and
# build/make/tools/releasetools/build_super_image.py emits ":none:" instead.
# So this file is damage limitation, not the recommended path.
#
#   Recommended:  bash work/scripts/42-flash.sh
#                 -- stays in bootloader fastboot, one `flash super`, never
#                    enters fastbootd, never touches recovery.
#
version 1
flash boot
flash init_boot
flash dtbo
flash vendor_boot
flash pvmfw
flash --apply-vbmeta vbmeta
flash vbmeta_system
reboot fastboot
update-super
flash odm
flash product
flash system
flash system_dlkm
flash system_ext
flash vendor
flash vendor_dlkm
if-wipe erase userdata
if-wipe erase metadata
flash recovery
